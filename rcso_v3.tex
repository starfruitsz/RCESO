\section{建模}\label{sec:model}
% =====================================================================
\subsection{被控对象与扰动分解}
\begin{equation}
  \dot y = b u+d,\qquad d = d_{\mathrm{step}}+d_{\mathrm{np}}+d_p,\qquad b=1,\ b_0=1.2\ (\text{失配}+20\%),
\end{equation}
$d_{\mathrm{np}}$ 为 $0.7\,$Hz 非周期分量；$d_p$ 为周期 $T$ 的谐波簇
$A[\sin\frac{2\pi t}{T}+0.4\sin\frac{6\pi t}{T}+0.2\sin\frac{10\pi t}{T}]$，$T=0.1\,$s（基波 $10\,$Hz）。
\textbf{“灰箱”的含义}：$d$ 中只有 $d_p$ 可重构，其余全划入总扰动由观测器兜底——结构已知的部分显式重建，未知部分用有限带宽观测器覆盖。
\subsection{低通重复控制（内模）通道}
\begin{equation}
  \tau\dot v+v = Q v(t-T)+\beta_3\st\!\Big(\frac{e_1}{\zeta}\Big)=:\chi,
  \qquad \ef := \chi-v = \tau\dot v .
  \label{eq:rc}
\end{equation}
\begin{itemize}
  \item $(\tau,Q)\to(0,1)$ 时 \eqref{eq:rc} 退化为经典重复控制，内模极点位于虚轴。\textbf{低通 $\tau$ 与遗忘因子 $Q$ 是把内模极点从虚轴拽入左半平面的两个旋钮}，代价是谐振峰有限宽——这正是 $(\tau,Q)$ 需要自适应的根源。
  \item 注入项有界（$|\st|\le1$），保证 $v$ 通道激励有界。
  \item $\ef=\tau\dot v$ 既是重复通道的内部失配量，也是\textbf{两个自适应通道唯一的驱动源}。
\end{itemize}
% =====================================================================
\subsection{低通 $\tau$ 的作用：为什么不能去掉}\label{sec:tau-role}
% =====================================================================
内模的低通 $\tau$ 常被误当成"可选的抗噪附件"；实际上它是\emph{内模回路的环路滤波器}。
把重复通道 \eqref{eq:rc} 写成算子形式：
\begin{equation}
v=\frac{1}{1+\tau s}\Bigl[Q\,v(t-T)+\beta_3\tanh\Bigl(\frac{e_1}{\zeta}\Bigr)\Bigr],
\label{eq:tau-loop}
\end{equation}
可见 $\tau$ 作用在\emph{旧记忆 $+$ 新注入}的\emph{和}上，处于回路内部，而不是对 $e_1$ 做前置滤波。

\paragraph{四个作用，均可量化。}
\begin{enumerate}
  \item \textbf{决定谐波处的泵浦增益}：$|G|_{\rm peak}=\beta_3/[\sqrt{1+(\omega\tau)^2}-Q]$。
        $\tau=20$\,ms、$10$\,Hz、$Q=0.98$ 时约 $6.4\times10^{3}$；
        而 $\tau\to0$ 时退化到 $\beta_3/(1-Q)=2\times10^{5}$，\textbf{高 $31$ 倍}。
  \item \textbf{决定相位滞后}，也就是 §\ref{sec:tes} 要补偿的量：$\delta^{*}=-\arctan(\omega\tau)/2\pi$；
        $\tau\to0$ 时 $\delta^{*}\to0$，T 自适应便失去补偿对象。
  \item \textbf{压低高频噪声的再循环}：$1$\,kHz 处增益 $\tau=20$\,ms 为 $32$、$\tau\to0$ 为 $2000$，\textbf{高 $63$ 倍}。
  \item \textbf{决定记忆的改写速率} $1/\tau$：一个周期改写 $99.3\%$、"回声只活一拍"都来自 $\tau=20$\,ms。
\end{enumerate}
此外 $\tau$ 把陷波由"深而窄"变为"浅而宽"：经典 RC（$\tau=0$）的陷波半宽约 $\pm0.3\%$，而 $\tau=20$\,ms 时谐振谷展宽——
这正是 §\ref{sec:tes-res} 中 $A(\delta)$ 平台很平（$-8\%$ 与 $-15\%$ 只差 $0.2$\,rpm）的原因。

\paragraph{"$e_1$ 已是带通，所以不需要低通"是误读。}
$e_1=y-z_1$ 是 ESO 的新息，其频谱是"未补偿的周期残余 $+$ 宽带噪声"；ESO 是低通估计器，并不带通。
真正呈带通/梳状特性的是\emph{回路}本身 $1/(1+\tau s-Qe^{-sT})$。因此"在 $e_1$ 上做带通"与"在回路里做低通"不是同一件事：
输入侧滤波不能替代环路滤波——回路在谐波处的峰高由 $\beta_3$、$Q$ 与 $\tau$ 共同决定。

\paragraph{去掉低通的实测后果。}
在 600\,rpm、$8$\,s、$2$\,s 加载 $0.50$ 额定、圆盘型线 V2 $+$ $5.9$\,Hz、实验室噪声、$f_{ci}=500$\,Hz 的口径下，
逐级减小 $\tau$、并按理论值重调周期 $\delta=-\arctan(\omega\tau)/2\pi$（单种子，脚本 \texttt{run\_tau\_sweep.py}）：

$\tau\le10$\,ms 起进入极限环：$\tau=20$\,ms 的稳态频谱无主峰（最大分量 $40$\,Hz、$6.8$\,rpm），

\paragraph{一个容易误判的陷阱。}
表中 $\tau=10$\,ms 的\emph{$10$\,Hz 残余}（$6.0$\,rpm）反而小于定稿（$14.1$\,rpm）——只看基波会得出"去掉低通更好"的错误结论；
但总 RMS 是 $3.7$ 倍、电流峰值 $1.5$ 倍，能量已转移到 $190$\,Hz。判据必须以\textbf{宽带 RMS 与电流峰值}为准，不能只看谐波幅值。

\paragraph{结论。}
\begin{enumerate}
  \item 保留一阶低通 $\tau_0=20$\,ms：它是稳定裕度与泵浦增益的设计变量，不是可选的抗噪件。
  \item 若目的是减小 $\delta^{*}$（少做周期修正），正确做法是采用\textbf{超前--滞后} $(1+\tau_z s)/(1+\tau_p s)$
        或线性相位 FIR，在保留高频衰减的同时压低 $f_0$ 处相位滞后；\emph{不是}去掉滤波。
  \item 若目的是更强的谐波抑制，应调整 $Q$ 与 $\beta_3$，并同时重算峰值增益与噪声增益——$\tau$、$Q$、$\beta_3$ 三者耦合。
  \item 边界：本表为单种子、$8$\,s 口径；换速与 $30$\,s 口径的结论见 §\ref{sec:limit}。
\end{enumerate}

\subsection{灰箱 ESO 与控制律}
\begin{equation}
  \dot z_1 = z_2+b_0u+v+\beta_1\,G(e_1;1,\delta,\lambda),\qquad
  \dot z_2 = \beta_2\,G(e_1;a,\delta,\lambda),\qquad
  \beta_1=2\omega_o,\ \beta_2=\omega_o^2,
\end{equation}
其中 $e_1=y-z_1$，校正增益
$G(e;a,\delta,\lambda)=\bigl(|e|+\delta\lambda^{-1/a}\bigr)^{a}\tanh\!\big(\lambda e/\delta\big)$，
$z_1$ 通道指数取 $1$、$z_2$ 通道指数取 $a<1$（默认 $a=0.75,\ \delta=0.10,\ \lambda=10$）；
取 $G(e)\equiv e$ 即退化为经典线性 ESO——\textbf{本文全部数值取类 fal 默认档（$a=0.75,\ \delta=0.10,\ \lambda=10$）}；线性档只作回归对照。
\begin{equation}
  u = \frac{u_0-z_2-v}{b_0},\qquad u_0 = k_p(r-y),\qquad e_1 = y-z_1 .
\end{equation}
与标准 ADRC 的唯一差别：$z_1$ 方程多出 $v$——周期分量已被\emph{预先补偿}，观测器只估计残余总扰动。

\begin{remark}[类 fal 增益的三条性质与两档口径]\label{rem:fal-gain}
式 (3) 的 $G$ 取何值就是\textbf{增益形状}本身：默认档 $a=0.75,\ \delta=0.10,\ \lambda=10$
（\texttt{eso\_law="fal"}，自 2026-10-22 起）；取 $G(e)\equiv e$ 为线性档
（\texttt{eso\_law="linear"}，可逐位复现 v4.0 的全部历史数据）。
三条性质决定它只宜作“默认 $+$ 对照”两档，而不是任意替换：
(i) 小信号斜率 $G'(0)=\delta^{a-1}$（与 $\lambda$ 无关），故等效观测带宽 $\omega_o^{\rm eff}=\omega_o\delta^{(a-1)/2}$，噪声本底随之变化；
(ii) 大信号 $G\to(|e|+\delta\lambda^{-1/a})^{a}\operatorname{sgn}e$：$a<1$ 次线性，$a=1$ 与线性同阶、只多一个宽 $\delta/\lambda$ 的偏移带；
(iii) $G$ 连续可导且 $\mathrm dG/\mathrm de>0$，但 \S\ref{sec:uub} 的 UUB 建立在 $\beta_1=2\omega_o,\beta_2=\omega_o^2$ 的\emph{线性}校正上（引理 2 的 $A_0$ 是常矩阵），换成 $G$ 后 $A_0$ 变成 $e_1$ 相关的扇区矩阵，\textbf{该界需重做，本文未做}。
\textbf{默认档的口径与代价（如实披露）}：类 fal 改变了 $e_1$ 的轨迹，故所有依赖 $e_1$ 的检测量（律 D 的门控、T 自适应的冻结判据）都必须在默认档下标定——本文全部数值即在该档下取得。
稳态代价依赖工作点与噪声档：600 rpm 实验室档的稳态 RMS 比线性档高 $19\%$，900/1200 rpm 反而低 $8\%/14\%$；报数必须同时写明 $(a,\delta,\lambda)$ 与噪声档。
\end{remark}

\section{自适应律（与实现逐条对应）}\label{sec:law}
\noindent\textbf{口径提示}：本节（内模侧自适应律）的数值来自\textbf{一阶标幺对象的机理实验}，只用于说明律的结构与性质；\textbf{真实逆变器对象}的性能与结论见 §\ref{sec:pmsm}--§\ref{sec:tes-res}，两者数字不可混引。\textbf{$Q$ 通道只作可选中性档}（§\ref{sec:bpw-result}）——最终推荐配置为“类 fal ESO + 重复通道 + 律 D + T 自适应”，不含 $Q$ 与任何事件触发律。

\subsection{辅助滤波与归一化基准（新定义）}\label{sec:aux}
辅助滤波给出归一化基准 $E_o$：取 $e_f^2$ 在最近一个扰动周期 $T$ 上的均值 $E_o(t)=\frac1T\int_{t-T}^{t}e_f^{2}(\tau)\,\mathrm d\tau$（实现为 \texttt{eo\_def="meansq"}）。归一化的对象因此是\emph{当前能量相对其周期均值}，而不是相对一个固定常数；$\varepsilon$ 只作分母下限（本设计 $\varepsilon=6.5$，必须是真实尺度，取 $10^{-6}$ 会让归一化损失分辨力）。
\subsection{归一化与 $\phi$ 压缩}\label{sec:x}
\begin{equation}
  \phi(x,c)=x-c\tanh\!\Big(\frac{x}{c}\Big),\qquad
  w(t)=\phi\!\Big(\frac{\ef^{2}(t)-E_o(t)}{E_o(t)+\varepsilon}, c\Big),
  \label{eq:phi-x}
\end{equation}
\textbf{实参取零均值形式} $\dfrac{\ef^{2}-E_o}{E_o+\varepsilon}$：分子减去自身的周期均值 $E_o$，使 $w$ 可正可负。\textbf{这一步是本版的关键}——若沿用 $\dfrac{\ef^{2}}{E_o+\varepsilon}$，由于 $\ef^{2}\ge0$ 恒成立，$w$ 恒正，其带通输出 $D$ 的均值系统性偏正，$Q$ 被单向压到底（实测 $Q$ 动作仅 $\pm1.5\times10^{-4}$，等于关死该通道）。改为零均值后 $\overline{D}\approx0$，$Q$ 才具备调节能力。
$\phi$ 把实参压到有限范围（$|x|\ll c$ 近似恒等，$|x|\gg c$ 饱和）；分母用 $E_o+\varepsilon$ 而非 $B_\kappa+\varepsilon$：$E_o$ 是分子的周期均值，尖峰瞬时 $\ef^{2}$ 可远大于 $E_o$。
\subsection{带通驱动（本版方案）}\label{sec:bpw}
把压缩后的 $w$ 分成\textbf{快、慢两路一阶低通}并取其差，得到\emph{带通}驱动：
\begin{equation}
  \dot w_f=\frac{w-w_f}{\tau_f^{w}},\qquad \dot w_l=\frac{w-w_l}{\tau_l^{w}},
  \qquad\Longrightarrow\qquad
  \boxed{\;D(t)=w_f(t)-w_l(t)\;}
  \label{eq:bpw}
\end{equation}
$w_f$、$w_l$ 由 \eqref{eq:phi-x} 的 $w$ 驱动；本设计 $\tau_f^{w}=5T$、$\tau_l^{w}=50T$。\textbf{带通的意义}：$w_f$ 先起、$w_l$ 后随，故 $D$ 对输入的\emph{频率与幅值变化}敏感，而在恒定输入下 $\to0$。
两条通道\textbf{共用同一个驱动} $D$，各自只保留\emph{独立}的 $(w_f,w_l)$ 状态：
\begin{equation}
  \boxed{\;D_Q(t)=\max\bigl(0, D(t)\bigr)\;}
  \label{eq:shared-D}
\end{equation}
即：\textbf{$Q$ 通道取\emph{单向死区} $\max(0,D)$}——只接受正向驱动、负向置零，故 $Q$ 只被单向下压、由回复项缓慢拉回 $1$（见 \S\ref{sec:qch}）。
选择单向死区的理由：$Q$ 的下界决定重复通道谐振增益 $\propto\beta_3/(1-Q)$，若允许负向驱动，$Q$ 会被推过 $1$ 而使该增益变号发散；单向死区同时给出\emph{单调}的下降—回归轨迹（实测 $Q$：$1.000\to0.386$ 后单调回升至 $1.000$，全程不过冲，见 \S\ref{sec:bpw-result}）。
\subsection{$Q$ 通道（遗忘因子自适应）}\label{sec:qch}
\begin{equation}
  \dot Q=-\frac{\mu}{2\tau_0}\bigl[1-(1-Q)^2\bigr]D_Q+\sigma_2\ath(1-Q),
  \label{eq:Q}
\end{equation}
\textbf{分母为 $2\tau_0$，不含 $\varepsilon$}（$\varepsilon$ 只出现在 $x$ 的分母里）。$Q$ 的定义域为 $(0,\,2)$。

$Q$ 的驱动取\textbf{单向死区} \eqref{eq:shared-D}：$D_Q=\max(0,D)$，只保留正向、负向置零。其效果见注 \ref{rem:one-sided} 与 \S\ref{sec:bpw-result} 的对比表。
\begin{remark}[$\sigma_2$ 决定 $Q$ 是否有动作]\label{rem:one-sided}
$\dot Q$ 的驱动与回复之比 $\propto\dfrac{\mu\,|D_Q|}{\sigma_2|{\ath}(1-Q)|}$，故 $\sigma_2$ 同时决定\textbf{降幅}与\textbf{回归速度}：

\textbf{方法上的教训}：早期把 $\sigma_2$ 调到 $10^{4}$ 使 $Q$ 降幅仅 $0.0053$（等于关死该通道），此时任何“自适应无收益”的结论都不成立——\emph{验证性能前必须先确认自适应状态确实在动作}。
\end{remark}

\begin{remark}[域限幅的事实描述（不作必要性断言）]\label{rem:proj-fact}
实测（$\sigma_2=5$、$q'_{\max}=1.0$）：模型侧 $Q'\in[0.2532,\,0.9991]$，未触界；状态 $Q\in(0,2)$ 亦未越界。
本版 $\tau\equiv\tau_0$ 固定，\textbf{不存在 $\tau$ 的状态侧/模型侧分岔问题}。
\textbf{故本注只陈述 $Q$ 通道的事实}：模型侧取限幅值、状态侧取宽域值，本工况下二者均未触界。
\end{remark}
\begin{equation}
  \chi = Q' v(t-T)+\beta_3\st\!\Big(\frac{e_1}{\zeta}\Big),\qquad \tau_0\dot v+v=\chi,\qquad
  Q'=\mathrm{clip}(Q,[0.2,1.0]) ,
  \label{eq:clip-model}
\end{equation}

即 $Q'$ 是\emph{唯一}进入被控对象与滤波时间常数的量，状态 $Q$ 只出现在自适应律内部（$\tau\equiv\tau_0$ 为常数）。
\begin{remark}[死区生效性的判别实验]\label{rem:deadzone-proof}
把模型侧死区收到 $q_{\max}=0.5$：若模型真用 $Q'$，状态应\emph{越过} $0.5$ 而不被截断。实测状态 $Q\in[0.0009,\,1.9966]$，其中 $Q>0.5$ 共 $\mathbf{122\,650}$ 步，而 $Q\ge2^{-}$ 为 $\mathbf{0}$ 步——同时证实：死区只作用于模型侧，状态只受定义域天花板约束。
\end{remark}
\begin{remark}[分解的效果：状态轨迹改变，控制效果不变]\label{rem:decouple-effect}
同激励同边界下三档的四窗统计\textbf{逐位相同}（(3)vs(2)：突变中 $+82.2\%$、恢复中 $+79.1\%$、突变中2 $-369.7\%$、恢复中2 $-200.2\%$、长期 $-349.1\%$），
差别只在状态 $Q$ 的实测区间（钳位 $[0.2000,1.8000]$，解耦 $[0.0009,1.9966]$）。\textbf{故 $(Q',\tau')$ 分解是表述层改进，不是性能杠杆}；它显式暴露了原结构被隐藏的事实：自由驱动的状态本来就会撞上定义域天花板。
\end{remark}

\subsection{最终方案的实测结果}\label{sec:bpw-result}
参数：驱动 \eqref{eq:bpw}，实参 \eqref{eq:phi-x}（零均值），$c=10^{-3}$，$\gamma=3$，$\sigma_1^{\mathrm{code}}=10^{5}$，$\mu=0.03$，$\sigma_2=5$，$\tau_f^{w}=5T$，$\tau_l^{w}=50T$，$k_p=10$，$\omega_o=40$，$Q_0=1.0$，模型侧上界 $q'_{\max}=1.0$。
工况：三段幅值突变（$t=0.5$ s 持续 $0.1$ s、$t=1.0$ s 持续 $0.5$ s、$t=2.0$ s 起持续），幅值倍数 $300$，$T_{\rm end}=10$ s。基准：\emph{无自适应}档（$Q\equiv1.0000$、$\tau\equiv\tau_0$ 恒定）。
\begin{remark}[$z_2$ 的直流分量为何必然存在]\label{rem:z2dc}
实测稳态（$5$--$10$ s）：$y=1.000001$、$u=-0.9049$、$z_2=1.0401$、$d=0.9021$。\textbf{$z_2$ 含有直流，这不是缺陷}：当 $b_0\ne b$ 时 ESO 估计的是\emph{总扰动}
\begin{equation}
  z_2\;\longrightarrow\;(b-b_0) u\;+\;d .
  \label{eq:z2dc}
\end{equation}
代入 $(b,b_0,u,d)=(1,\,1.2,\,-0.9049,\,0.9021)$ 得 $z_2\to1.0831$，与实测 $1.0401$ 吻合（残差来自有限观测带宽 $\omega_o$）。
\textbf{该直流是跟踪所必需的}：控制律 $u=\dfrac{k_p(r-y)-z_2-v}{b_0}$ 要求 $z_2$ 恰好抵消 $d$ 的直流分量，否则 $y$ 必含稳态误差；实测 $|1-y|=10^{-6}$ 证明其工作正常。
\textbf{来源可逐项核对}：(i) 阶跃扰动自身的直流；(ii) 模型失配项 $(b-b_0)u$。把任一项人为置零，$z_2$ 的直流随之消失，但 $y$ 会出现稳态偏差。
\end{remark}

\end{remark}

\section{律 D：比例增益与观测带宽的联合调度}\label{sec:lawd}
% =====================================================================
\noindent\textbf{定位}：律 D 是本报告唯一的“突变侧”自适应律（2026-11-11 定稿，取代只调 $k_p$ 的旧版）。
它与 §\ref{sec:tes} 的 T 自适应分工明确：律 D 的时间常数约 $20$\,ms，专管加载/阶跃等\emph{非周期}突变；
T 自适应的时间常数为 $1.5$--$4.5$\,s，专管\emph{周期}扰动的相位失配。两者相差两个数量级，可同时投运。

\subsection{设计与实现}\label{sec:lawd-design}
检测量取跟踪误差在扰动周期 $T$ 上的滑动均值：
\begin{equation}
\bar e_1(t)=\frac1T\int_{t-T}^{t}e_1(\tau)\,\mathrm d\tau,\qquad
g_{\rm fire}(t)=\frac{\bar e_1^2(t)}{\bar e_1^2(t)+\varepsilon_g^2},
\label{eq:lawd-gate}
\end{equation}
调度律同时抬高比例增益与观测器带宽（$\beta_1,\beta_2$ 按二阶带宽一致缩放）：
\begin{equation}
k_p^{\rm eff}=k_p\bigl(1+k_b\,g_{\rm fire}\bigr),\qquad
\omega_o^{\rm eff}=\omega_o\bigl(1+a\,g_{\rm fire}\bigr),\qquad
\beta_1'=2\omega_o^{\rm eff},\quad \beta_2'=(\omega_o^{\rm eff})^{2}.
\label{eq:lawd-eff}
\end{equation}
闭环实现为
\begin{equation}
u=\frac{1}{b_0}\Bigl[k_p^{\rm eff}(r-y)-z_2-v\Bigr],\qquad
\dot z_1=z_2+b_0u+v+\beta_1'\,G(e_1),\qquad
\dot z_2=\beta_2'\,G(e_1),
\label{eq:lawd-law}
\end{equation}
内模通道 $\tau_0\dot v+v=Q_0v(t-T)+\beta_3\tanh(e_1/\zeta)$ 不参与调度（$\beta_3$ 自适应实测无收益，见 §\ref{sec:lawd-b3}）。
定稿参数：$k_b=8$、$a=2$、$\varepsilon_g=0.01$。

\subsection{为什么用 $e_1$ 的 $T$ 窗均值}\label{sec:lawd-why}
\begin{enumerate}
  \item \textbf{对周期分量结构性失明}：任意 $k\cdot f_0$ 分量在长度 $T$ 的窗口上均值为零，故 $\bar e_1$ 不含内模正在抑制的谐波，
        $g_{\rm fire}$ 只对“非 $1/T$ 栅格”的分量响应（加载阶跃、$0.7$\,Hz 非周期负载、$5.9$\,Hz 振动）。
        形式化陈述与证明见附件（链引理：$e_1=p+q$、$p$ 为 $T$ 周期零均值 $\Rightarrow \bar e_1\equiv\bar q$）。
  \item \textbf{空载段逐位退回定稿}：无周期、无非周期扰动时 $\bar e_1\equiv0$、$g_{\rm fire}\equiv0$，
        于是 $k_p^{\rm eff}\equiv k_p$、$\omega_o^{\rm eff}\equiv\omega_o$——不放大噪声本底。
        这是“仅提 $k_p$”的旧方案做不到的（旧方案全局生效，必然抬高噪声）。
  \item \textbf{零额外传感}：检测量只用到控制律已有的 $e_1$，不需要负载观测器或外部事件信号。
\end{enumerate}
\begin{remark}[检测器的真实工作区间]\label{rem:lawd-range}
$g_{\rm fire}$ 同时对 $0.7$\,Hz 非周期负载、$5.9$\,Hz 振动与加载阶跃响应（均不在 $1/T$ 栅格上），
故它并非“只在突变瞬间动作”，而是“非周期误差越大、增益越高”的连续调度：空载段 $g\equiv0$（逐位退回定稿），
加载后 $g$ 持续非零。稳态段的这部分增益正是“稳态段同时改善”的来源；
也说明 $\varepsilon_g$ 的工作点落在检测器拐点上，必须按对象实际非周期误差水平标定（对比见实验记录 2026-11-11 §10.4--10.5）。
\end{remark}
\begin{remark}[为何不把 $\phi$ 压缩放在检测器前]\label{rem:lawd-phi}
$T$ 窗平均已把宽带噪声压低 $\sqrt{T/\Delta t}\approx70$ 倍（空载段 $g\equiv0$），没有噪声泄漏需要修；
而 $\phi$ 压缩的是大值，恰是待检测的突变信号。实测 $c$ 由 $0.002$ 增到 $0.05$ 时，加载瞬态与稳态 RMS 均单调劣化、无改善区间。
\end{remark}

\section{T 自适应：标称周期 + 锁相极值搜索}\label{sec:tes}
% =====================================================================
\noindent\textbf{问题}：工程上取 $T=2\pi/\omega$ 只保证\emph{频率}对齐，不保证\emph{相位}对齐——
记忆低通 $\tau_0$、控制拍离散化、电流环与采样延时共同引入等效相移，使真实最优周期系统性偏离标称值。
本节在既有频率自适应之外，增加一层以基波残余幅值 $A$ 为目标的\textbf{锁相极值搜索（ESC）}在线修正相位。

\subsection{失配机理与相位对齐条件}\label{sec:tes-phase}
内模对频率 $f$ 的抑制能力由分母模长决定：
\begin{equation}
\bigl|\mathcal G(f)\bigr|\propto\frac{1}{\bigl|1-Qe^{-j2\pi fT}\bigr|},
\qquad
\bigl|D(j\omega)\bigr|^{2}=1+(\omega\tau)^{2}+Q^{2}-2Q\bigl[\cos\theta-\omega\tau\sin\theta\bigr],\quad \theta=\omega T .
\label{eq:tes-D}
\end{equation}
即 $1\%$ 的周期误差（相位 $3.6^{\circ}$）使有效增益衰减 $3.3$ 倍，$5\%$ 时衰减 $15$ 倍；陷波半宽仅 $\pm0.3\%$。
\begin{proposition}[相位对齐条件]\label{prop:tes-phase}
对任意 $Q\in(0,1)$、$\tau>0$，$|D(j\omega)|$ 关于 $\theta$ 的唯一极小点为 $\omega T=2\pi k-\arctan(\omega\tau)$；
当 $T=T_g(1+\delta)$、$T_g=2\pi/\omega$ 且取 $k=1$ 时，最优周期修正为
\begin{equation}
\delta^{*}(\omega)=-\frac{\arctan(\omega\tau)}{2\pi}.
\label{eq:tes-dstar}
\end{equation}
极小值 $|D|_{\min}=\sqrt{1+(\omega\tau)^{2}}-Q>0$：内模不是理想无穷增益，抑制深度仍受 $Q$ 限制。
\end{proposition}
\begin{proof}
最小化 $|D|^{2}$ 等价于最大化 $f(\theta)=\cos\theta-\omega\tau\sin\theta=\sqrt{1+(\omega\tau)^{2}}\cos(\theta+\varphi)$，$\varphi=\arctan(\omega\tau)$；
其唯一最大值在 $\theta+\varphi=2\pi k$ 处；代入 $T=T_g(1+\delta)$ 与 $\omega T_g=2\pi$ 即得。
\end{proof}
\begin{remark}[转速依赖性]\label{rem:tes-speed}
$\delta^{*}(\omega)$ 随转速单调漂移：$\omega\to0$ 时趋于 $0$，$\omega\to\infty$ 时趋于 $-25\%$。
$\tau=20$\,ms 时 600\,rpm 为 $-14.3\%$、300\,rpm 为 $-8.9\%$——\textbf{换速后必然重新失配}，必须在线整定或前馈重置。
\end{remark}

\subsection{基波幅值观测器}\label{sec:tes-obs}
取 $\pm10\%$ 带通（状态变量结构，$Q_{\rm bp}=5$），再取一周期均方根：
\begin{equation}
A[k]=\sqrt{\frac{1}{N_T}\sum_{i=0}^{N_T-1}y_{\rm bp}^{2}[k-i]},\qquad N_T=\frac{T_g}{\Delta t_s}.
\label{eq:tes-amp}
\end{equation}
\begin{remark}[$A$ 只能做目标，不能做调度量]\label{rem:tes-amp}
$A$ 的物理含义是“内模频段内尚未被抑制的残余”，与抑制质量成反比，故只能作\emph{最小化}目标。
实测用它调度 $\beta_3$ 会使空载噪声上升 $11\%\sim55\%$、加载跌落反而加深：$A$ 小恰是控制有效的\emph{结果}，
用它降增益等于把残余顶在参考值上。
\end{remark}

\subsection{锁相极值搜索}\label{sec:tes-esc}
两层周期：内层给频率、外层修相位。
\begin{equation}
T[k]=T_{\rm gross}[k]\bigl(1+\delta[k]\bigr),\qquad
\delta[k]=\hat\delta_k+\Delta\sin(\omega_d t),\qquad |\hat\delta_k|\le\delta_{\max},
\label{eq:tes-two-layer}
\end{equation}
\begin{equation}
I_k=\frac{2}{NP_d}\int_{(k-1)NP_d}^{kNP_d}A(t)\sin(\omega_d t)\,\mathrm dt,
\qquad
\hat\delta_{k+1}=\mathrm{clip}\bigl(\hat\delta_k-\sigma\,\mathrm{sgn}(I_k),\ \pm\delta_{\max}\bigr).
\label{eq:tes-esc}
\end{equation}
\begin{remark}[收敛条件]\label{rem:tes-conv}
平台附近可二次近似 $A(\delta)\approx A^{*}\bigl(1+\kappa(\delta-\delta^{*})^{2}\bigr)$，解调量
$I_k\approx2\kappa A^{*}(\hat\delta_k-\delta^{*})\Delta$。可分辨条件为单步调制量大于 $A$ 的噪声：
$2\kappa A^{*}\sigma|\hat\delta-\delta^{*}|\gtrsim\eta_A$。实测 $\kappa\sim20$--$130$、$\Delta=0.5\%$：
偏离最优点 $1\%$ 以上即可可靠定向，$1\%$ 以内停在线性环内，代价约 $0.2\%$ 的 $A$。
\end{remark}

\subsection{离散实现}\label{sec:tes-impl}
\begin{itemize}
  \item \textbf{分数阶延迟}：$n_d=T/\Delta t_s$ 取小数，$v[k-n_d]\approx(1-\lambda)v[k-i_0]+\lambda v[k-i_0-1]$，$i_0=\lfloor n_d\rfloor$。
        不插值时 $1$\,ms 量化对应约 $1\%$ 的 $T$，会吞掉 $1\%$ 的搜索步长。
  \item \textbf{冻结与保护}：标称周期突变 $>5\%$ 或 $|1-y|>0.1$ 时冻结 ESC，保持 $\hat\delta$ 共 $2$\,s。
  \item \textbf{前馈重置}：冻结期间每毫秒按实测转速重设
        $\hat\delta\leftarrow\mathrm{clip}\bigl(-\arctan(\omega_f\tau)/2\pi,\ \pm\delta_{\max}\bigr)$，解冻后 ESC 只做 $\pm3\%$ 的细调。
  \item \textbf{终版参数}：$\delta_{\max}=\pm20\%$、$\sigma=1\%$、$\Delta=0.5\%$、$P_d=1.5$\,s、$N=3$、
        $\hat\delta_0=-14.3\%$（理论初值）、死区 $I_{\rm dead}=10^{-5}$。
\end{itemize}
\begin{remark}[与律 D 的时间尺度分离]\label{rem:tes-lawd}
ESC 的更新周期 $NP_d=4.5$\,s，律 D 的门控时间常数约 $20$\,ms，二者相差两个数量级以上：
律 D 在 ESC 的一个更新周期内已完成多次动作，ESC 看到的 $A$ 是“律 D 稳定后”的残余。
故两者在频域上互不干扰，§\ref{sec:tes-res} 的实测即两者同时开启。
\end{remark}

\subsection{附：T 自适应的直观推导}\label{sec:tes-intuition}
% =====================================================================
本小节把 §\ref{sec:tes-phase}--§\ref{sec:tes-impl} 的结论按“先机理、后证明”重讲一遍，目标是让第一次读的人能回答三个问题：
标称周期差在哪里、最优周期为什么恰好是那个数、ESC 为什么能找到它。形式化证明见附件 \texttt{proof\_lawD\_T} 第 4 节。

\paragraph{(1) 标称周期差在哪里。}
内模用 $v(t-T)$ 取出的记忆，必须与当前扰动\textbf{同相}才能相消；而记忆是经时间常数 $\tau$ 的低通写入的，
在频率 $\omega$ 处的相位滞后为
\begin{equation}
\varphi=\arctan(\omega\tau).
\label{eq:intuition-phi}
\end{equation}
600 rpm（$10$ Hz）、$\tau=20$\,ms 时为 $51.5^{\circ}$，折合 $14.3$\,ms。与此同时内模的陷波极窄：$1\%$ 周期误差（相位 $3.6^{\circ}$）
使有效增益衰减 $3.3$ 倍、$5\%$ 衰减 $15$ 倍，半宽仅 $\pm0.3\%$。两者叠加的后果是\textbf{标称周期下陷波没有对准扰动}，
这是 §\ref{sec:tes-res} 中不整定时稳态残余 $30.54$\,rpm 的主要来源。

\paragraph{(2) 最优周期为什么是 $-\arctan(\omega\tau)/2\pi$。}
把内模写成传递函数 $V/W=1/(1+\tau s-Qe^{-sT})$，抑制能力由分母在 $s=j\omega$ 处的模长决定。记 $\theta=\omega T$，有
\begin{equation}
|D(j\omega)|^{2}=1+(\omega\tau)^{2}+Q^{2}-2Q\bigl[\cos\theta-\omega\tau\sin\theta\bigr]
=1+(\omega\tau)^{2}+Q^{2}-2Q\sqrt{1+(\omega\tau)^{2}}\cos(\theta+\varphi),
\label{eq:intuition-D2}
\end{equation}
其中 $\varphi=\arctan(\omega\tau)$。$\cos$ 的唯一极大点在 $\theta+\varphi=2\pi k$，即 $\omega T=2\pi k-\arctan(\omega\tau)$；
代入 $T=T_g(1+\delta)$、$\omega T_g=2\pi$ 并取 $k=1$，即得命题~\ref{prop:tes-phase} 的 $\delta^{*}=-\arctan(\omega\tau)/2\pi$。
\textbf{物理读法}：把周期缩短 $\arctan(\omega\tau)/\omega$ 这么长的时间，正好抵消低通的相位滞后——600 rpm 是 $14.3$\,ms/$100$\,ms，
故 $\delta^{*}=-14.3\%$；300 rpm（$5$ Hz）是 $17.9$\,ms/$200$\,ms，故 $-8.9\%$。
极小值 $|D|_{\min}=\sqrt{1+(\omega\tau)^{2}}-Q>0$：\textbf{相位对齐只把残余压到最小，不会压到零}，因为 $Q<1$。

\paragraph{(3) ESC 为什么能找到它。}
最优附近 $A(\delta)\approx A^{*}\bigl[1+\kappa(\delta-\delta^{*})^{2}\bigr]$ 是一个“碗”。加正弦抖动 $\delta=\hat\delta+\Delta\sin(\omega_d t)$ 代入，
乘 $\sin(\omega_d t)$ 后对整数个抖动周期平均，常数项与 $\sin^{2}$ 项都被平均掉，只剩
\begin{equation}
I_k\approx2\kappa A^{*}\,(\hat\delta_k-\delta^{*})\,\Delta ,
\label{eq:intuition-I}
\end{equation}
\textbf{其符号直接指示现在站在碗的哪一侧}；于是 $\hat\delta_{k+1}=\hat\delta_k-\sigma\,\mathrm{sgn}(I_k)$ 就是“往低处走一步”。
三个参数各管一件事：$\Delta$ 管信噪比与稳态抖动（代价正比于 $\kappa A^{*}\Delta^{2}$）、$\sigma$ 管步长与收敛速度、$N$ 管解调积分长度。
可分辨条件是 $2\kappa A^{*}\sigma|\hat\delta-\delta^{*}|\gtrsim\eta_A$：偏离最优点 $1\%$ 以上才可靠定向，$1\%$ 以内靠死区停住。

\paragraph{(4) 稳定性为什么归结为 $\mu<1-Q^{2}$。}
证明只有三步，每步一句话：
\begin{enumerate}
  \item \textbf{时变延迟的能量增益}：$(\mathcal D_T v)(t)=v\bigl(t-T(t)\bigr)$ 在 $|\dot T|\le\mu$ 下满足
        $\|\mathcal D_T v\|_2^{2}\le\|v\|_2^{2}/(1-\mu)$——变量替换 $s=t-T(t)$，时间轴被拉伸的倍率有界；
  \item \textbf{内模回路增益}：内模是 $v=\mathcal H\bigl(w+Q\mathcal D_T v\bigr)$，一阶低通满足 $\|\mathcal H\|\le1$，故
        $\|v\|\le\|w\|+Q\|v\|/\sqrt{1-\mu}$，环路增益即 $\gamma=Q/\sqrt{1-\mu}$；
  \item \textbf{小增益}：$\gamma<1$ 等价于 $\mu<1-Q^{2}$。$Q=0.98$ 时 $\mu_{\max}=0.0396$——\textbf{周期每秒最多允许变化约 $4\%$}。
\end{enumerate}
这条条件把“遗忘因子”与“周期变化率”绑在一起：$Q\to1$ 时内模趋于临界稳定、$\mu_{\max}\to0$，周期稍有变化就失稳。
数字体检：ESC 准静态每 $4.5$\,s 走 $1\%$（约 $0.22\%/$s），离 $3.96\%/$s 很远，定理适用；
换速 $600\to300$ 时 $|\dot T|\approx1$，明确越界，只能靠\textbf{冻结判据}（标称周期突变 $>5\%$ 即冻结 $2$\,s）与“解对 $T$ 连续依赖”兜底——
这也是 §\ref{sec:limit} (P5) 如实披露的边界。测速滤波的设计约束 $\tau_f\ge T|\dot\omega|/(\omega\mu_{\max})$ 在 600 rpm 阶跃下给出 $\ge0.05$\,s，
定稿取 $\tau_f=50$\,ms。

\paragraph{(5) 四个容易读错的地方。}
\begin{enumerate}
  \item \textbf{“公式说 $-14.3\%$，实测为何 $-12\%$”}：公式只含记忆低通那 $20$\,ms；控制拍离散化、电流环与测速滤波还叠加 $2$--$3$ 个百分点的附加相位，
        故实测谷在 $-12\%$（300 rpm 实测 $-8\%$、理论 $-8.9\%$）：趋势一致、绝对值有偏。
  \item \textbf{“$A$ 最优在 $-12\%$、RMS 最优为何在 $-15\%$”}：$A$ 只含基波，RMS 还含 $5.9$\,Hz 振动与噪声；整定看 $A$、验收看 RMS，两者不可混读。
  \item \textbf{“为什么不直接减小 $\tau$ 或提高 $Q$”}：$\tau$ 是陷波宽度与噪声抑制的折中，而 $Q\to1$ 会使 $\mu_{\max}=1-Q^{2}\to0$；
        本文的做法是\textbf{在给定 $(\tau,Q)$ 下补偿相位}，而不是取消低通。
  \item \textbf{“换速时定理还成立吗”}：不成立。准静态有定理，换速瞬态只有“有界、可重复”的实验证据与冻结判据，未给出瞬态期间的定量界。
\end{enumerate}

\section{可证结论：驱动有界性与名义闭环 UUB}\label{sec:stab}
\subsection{结构前提：驱动有界性}\label{sec:newpremise}
本版驱动 $D=w_f-w_l$（\eqref{eq:bpw}）是\emph{带通}，其“非负 / 周期零均值”均不成立，故不能沿用旧放缩。下面两条引理给出对本版驱动直接成立的前提，\textbf{证明只用到 $\phi$ 的 $1$-Lipschitz 性与滤波核为凸组合}，与驱动的符号结构无关。
\begin{lemma}[零均值带通驱动的有界性]\label{lem:Dbound}
设 $M_t:=\sup_{s\le t}\ef^{2}(s)<\infty$、$\varepsilon>0$、$c>0$、$\tau_f^{w},\tau_l^{w}>0$，实参取零均值形式 \eqref{eq:phi-x}。则
\begin{equation}
  |D(t)|\le\min\Bigl\{2,\;\frac{2M_t}{\varepsilon}\Bigr\},\qquad\text{从而}\qquad
  |D_Q(t)|=|D(t)|.
  \label{eq:Dbound}
\end{equation}
\emph{证明.} 记 $x=\dfrac{\ef^{2}-E_o}{E_o+\varepsilon}$，则 $\phi(0)=0$ 且 $\phi(\cdot,c)$ 为 $1$-Lipschitz，故 $|w|=|\phi(x,c)|\le|x|$。
又 $|x|\le\dfrac{|\ef^{2}|+E_o}{E_o+\varepsilon}\le1+\dfrac{|\ef^{2}|}{E_o+\varepsilon}\le1+\dfrac{M_t}{\varepsilon}$；
且 $|x|\le\dfrac{|\ef^{2}-E_o|}{\varepsilon}\le\dfrac{M_t}{\varepsilon}$（用 $E_o\ge0$）。两式取小即得 $|w|\le\min\{1,M_t/\varepsilon\}$。
一阶低通的核 $\frac{1}{\tau^{w}}e^{-t/\tau^{w}}$ 非负、积分为 $1$，故 $w_f,w_l$ 均为 $w$ 的历史凸组合，各自落在 $[\inf w,\sup w]$ 内，于是 $|D|=|w_f-w_l|\le2\sup|w|$。
因 $D_Q=D$（\eqref{eq:shared-D}），驱动界即为 $Q$ 通道的界。$\square$
\end{lemma}
\begin{remark}[零均值实参在分析上的两重好处]\label{rem:zero-mean}
\textbf{(i) 界更紧且与 $\varepsilon$ 解耦}：实参取 $\ef^{2}/(E_o+\varepsilon)$（恒正）时只能得到 $|D|\le2M_t/\varepsilon$，$\varepsilon\to0$ 时界发散；取零均值形式后有 $|D|\le2$，\textbf{对一切 $\varepsilon>0$ 一致成立}。
\textbf{(ii) 平衡点回到可用区}：恒正实参使 $w$ 恒正、$\overline{D}>0$，$Q$ 被单向压到底（实测动作仅 $\pm1.5\times10^{-4}$）；零均值实参使 $\overline{D}\approx0$，$Q$ 的平衡点回到 $1$ 附近，$Q$ 才具备调节能力。
\end{remark}
\begin{remark}[闭环一致有界性的归并与仍开放的问题]\label{rem:robust}
\textbf{(已归并)} 名义闭环的 UUB 与“含 $Q$ 与律 D（$k_p,\omega_o$）调度”的推论见 §\ref{sec:uub}：$v$ 通道自持有界（界 $\beta_3/(1-Q_0)$，与 $e_1$ 大小无关），
ESO 误差与跟踪误差的界显式依赖 $(\beta_3,Q_0,\omega_o,k_p,b,b_0,H)$；且因 $g\in[0,1)$、$\kappa\ge0$ 使增益只增不减，\textbf{名义界即调度闭环的界}。
\textbf{(仍开放)} (i) 随机噪声（电流/负载噪声）不属该确定性框架，其定量影响由敏感性一节单独量化；
(ii) 三个界都是\emph{保守上界}（与仿真实测相差 $10^{2}\sim10^{5}$ 倍），收紧界需更强的结构信息。
\end{remark}
\subsection{名义闭环的一致有界性（UUB）：完整证明}\label{sec:uub}
\textbf{设定}：名义闭环指 $\tau\equiv\tau_0$、$Q\equiv Q_0$ 固定、调度全关（$g\equiv0$）的闭环。
记 $e_1=x_1-z_1$、$e_2=x_2-z_2$，$x_2$ 为含输入失配的扩张状态，$\dot x_2=:h$。
\textbf{假设}：(A1) $|h(t)|\le H$；(A2) $b,b_0,k_p,\omega_o,\tau_0,\zeta>0$、$Q_0\in(0,1)$、$\beta_3\ge0$；(A3) $r$ 为常数、$d$ 有界。

\noindent\textbf{引理 1（重复通道自持 UUB）}：由 $|\tanh(\cdot)|\le1$，重复通道输入满足
$|Q_0v(t-T)+\beta_3\tanh(e_1/\zeta)|\le Q_0\bar V+\beta_3$（$\bar V:=\sup_{[t-T,t]}|v|$）；
一阶低通在 $|w|\le W$ 下满足 $|v(t)|\le W+(|v(0)|-W)e^{-t/\tau_0}$，逐周期迭代得
\begin{equation}
\limsup_{t\to\infty}|v(t)|\le\bar v:=\frac{\beta_3}{1-Q_0}.
\label{eq:uub-v}
\end{equation}
\emph{关键}：该界\textbf{与 $e_1$ 的大小无关}（因 $\tanh$ 有界），故 $v$ 通道“自持”，不构成代数环。

\noindent\textbf{引理 2（ESO 误差 UUB）}：ESO 误差动态为 $\dot e_1=-2\omega_o e_1+e_2-v$、$\dot e_2=-\omega_o^{2}e_1+h$。
取尺度化坐标 $\xi=[e_1,\ e_2/\omega_o]^{\!\top}$，则 $\dot\xi=\omega_o A_0\xi+w_d$，
$A_0=\left[\begin{smallmatrix}-2&1\\-1&0\end{smallmatrix}\right]$（特征值 $-1$ 二重），$w_d=[-v,\ h/\omega_o]^{\!\top}$。
解 $A_0^{\!\top}P_0+P_0A_0=-I$ 得 $P_0=\left[\begin{smallmatrix}0.5&-0.5\\-0.5&1.5\end{smallmatrix}\right]\succ0$（特征值 $0.191/1.809$），$\|P_0B_0\|_F=\sqrt3$。
取 $V=\xi^{\!\top}P_0\xi$ 得 $\dot V\le-\omega_o\|\xi\|^{2}+2\sqrt3\,\|\xi\|\,\|w_d\|$，故
\begin{equation}
\limsup_{t\to\infty}\|\xi(t)\|\le\frac{2\sqrt3}{\omega_o}\sqrt{\bar v^{2}+\bigl(H/\omega_o\bigr)^{2}} .
\label{eq:uub-eso}
\end{equation}

\noindent\textbf{引理 3（跟踪误差 UUB）}：令 $\sigma=r-y$，由 $\dot y=\frac{b}{b_0}(k_p\sigma-z_2-v)+d$ 得
$\sigma\dot\sigma\le-\frac{bk_p}{b_0}\sigma^{2}+|\sigma|\bigl(|d|+\frac{b}{b_0}(|z_2|+\bar v)\bigr)$，而 $z_2=x_2-e_2$ 有界，故
\begin{equation}
\limsup_{t\to\infty}|\sigma(t)|\le\frac{b_0}{bk_p}\Bigl(\sup|d|+\frac{b}{b_0}\bigl(\sup|x_2|+\bar v\bigr)\Bigr).
\label{eq:uub-track}
\end{equation}

\noindent\textbf{定理（名义闭环 UUB）}：在 (A1)--(A3) 下，式 \eqref{eq:uub-v}--\eqref{eq:uub-track} 同时成立，
闭环全部信号一致最终有界，界\textbf{显式}依赖 $(\beta_3,Q_0,\omega_o,k_p,b,b_0,H)$。
\emph{注}：(a) 本例回代：$\beta_3=4000$、$Q_0=0.98\Rightarrow\bar v=2\times10^{5}$、$\omega_o=500$；三个界都是\textbf{保守上界}
（仿真实测 $|v|\approx4.5\times10^{2}$、$|e_1|\approx0.05$，相差 $10^{2}\sim10^{5}$ 倍），定量行为以仿真为准；
(b) (A1) 的 $h$ 含 $\dot u$ 项，而 $u$ 由有界的 $z_2,v$ 构成，故 $h$ 的有界性由前述界隐式给出（有限）。

\noindent\textbf{推论（含 $Q$ 与律 D 调度）}：$g=\bar e_1^{2}/(\bar e_1^{2}+\varepsilon_g^{2})\in[0,1)$ 由构造保证，$k_b,a\ge0$，
故 $k_p^{\rm eff}=k_p(1+k_b g)\ge k_p$、$\omega_o^{\rm eff}=\omega_o(1+a g)\ge\omega_o$；
式 \eqref{eq:uub-eso}\eqref{eq:uub-track} 的界对 $(\omega_o,k_p)$ \textbf{单调不增}，故\emph{名义界即调度闭环的界}：
\textbf{调度可证地不放大一致最终界}。这与“调度放大噪声本底”的仿真事实不矛盾——噪声不在本确定性框架内，由敏感性一节单独量化。

\subsection{$\beta_3$ 不参与调度}\label{sec:lawd-b3}
把 $\beta_3$ 也按同一门控调度（$b>0$）只会推高电流：$b=1$ 时 RMS $38.2$（对比 $b=0$ 的 $38.6$，几乎不变）而

\subsection{机理}\label{sec:lawd-mech}
抬高 ESO 带宽 $\Rightarrow$ 扰动估计更快 $\Rightarrow$ 更早出力；因此\textbf{同样的瞬态改善只需更小的电流}。
这解释了为什么“仅提 $k_p$”会把电流顶到 $7.11$\,A，而“$k_p+\omega_o$ 一起提”能压回额定 $6.31$\,A：
前者靠反馈硬顶，后者靠估计前移。900 rpm 上甚至出现“瞬态更好、电流更小”（$5.89\to5.53$\,A）的双赢，
这正是“用信息（$T$ 窗均值）而不是用增益”的收益。

\section{适用范围之外（如实披露）}\label{sec:limit}

\subsection{复位型律 F 已作废（2026-11-12）}\label{sec:no-lawf}
本报告 v4.0 及以前版本含“状态复位律 F”（F-A 前馈 oracle、F-B 记忆失配 $D_m+g$ 触发、反相复位 $\alpha=-1$）。
该路线\textbf{整体撤除}，不再作为正文结论。理由是它在时间尺度上不成立：
\begin{enumerate}
  \item \textbf{回声只活一拍}：$\tau_0=20$\,ms 的记忆以 $1$\,ms 一拍向新误差靠拢，一个周期（$100$\,ms）内旧记忆被改写
        $1-0.951^{100}\approx99.3\%$；而任何“检测--复位”至少需要 $0.5$--$1$ 个周期做判定。
  \item \textbf{oracle 上界也很低}：事件时刻精确已知（零延迟）时，首周期回声的可见度上界只有 $1.82$ 倍，延迟 $10$\,ms 收益即归零。
  \item \textbf{判据在真实对象上不可分}：包络比值判据在事件期只到 $1.32$（与加载段 $1.3\sim1.5$ 重叠）；
        记忆失配判据可触发但依赖阈值与相位精度；反相事件需 $\alpha=-1$ 的方向判别，未证明、未接入。
\end{enumerate}
\noindent 结论：回声抑制需要\textbf{外部事件信号或更低延迟的相位估计}，不在本报告范围内；律 F 的脚本与数据只作历史证据保留，\textbf{不得用于新结果}。
保留的机理结论：$\kappa=\beta_3T/\tau_0$ 决定回声寿命，$A_g/A\approx25/\kappa$；$Q$ 属参数级遗忘，时间尺度 $10^{2}\sim10^{6}T$，追不上 $0.1$--$0.2T$ 的回声。

\section{结论}\label{sec:concl}
% =====================================================================
\begin{enumerate}
  \item \textbf{结构}：真实对象上为“低通重复通道（内模）+ 灰箱 ESO + 律 D + T 自适应”。
        三条回路共用 $e_f=\tau\dot v$ 或 $e_1$，不能分别证明，必须整体处理。
  \item \textbf{律 D（改良版）}：检测量取 $e_1$ 的 $T$ 窗均值，对任何 $k f_0$ 分量结构性失明，故只对非周期事件响应，且空载段逐位退回定稿。
        在电流约束 $|i_q|\le6.31$\,A 下，$k_p$ 与 $\omega_o$ 联合调度（$k_b=8,a=2$）在 600 rpm 给出
        突变窗 RMS $-23.2\%$、恢复时间 $-53\%$、跌落 $-7.1\%$、稳态持平；900 rpm 复核 $-14.3\%$、恢复 $43\to7$\,ms、电流反降 $6\%$。
  \item \textbf{T 自适应}：标称周期 $2\pi/\omega$ 只对齐频率、不对齐相位；相位对齐条件给出
        $\delta^{*}(\omega)=-\arctan(\omega\tau)/2\pi$（600 rpm 为 $-14.3\%$、300 rpm 为 $-8.9\%$），实测最优 $-12\%$ 与 $-8\%$。
        锁相 ESC 把稳态 RMS 由 $30.54$ 降到 $27.72$\,rpm（$-9.2\%$），与离线最优只差 $0.16$\,rpm；
        换速时前馈重置把 $25.32$ 降到 $19.83$，往返试验完全可逆。
  \item \textbf{一致有界性}：$v$ 通道自持 UUB（界 $\beta_3/(1-Q_0)$，与 $e_1$ 无关）；ESO 误差界
        $\frac{2\sqrt3}{\omega_o}\sqrt{\bar v^{2}+(H/\omega_o)^{2}}$；跟踪误差界 $\frac{b_0}{bk_p}(\sup|d|+\frac{b}{b_0}(\sup|x_2|+\bar v))$。
        因 $g\in[0,1)$ 且调度只增不减增益，\textbf{名义界即调度闭环的界}——律 D 不放大 UUB 界。
        T 自适应在准静态下满足小增益 $\gamma=Q/\sqrt{1-\mu}<1$；换速瞬态由冻结判据兜底。
  \item \textbf{律 F 作废}：复位型律 F（含 F-A/F-B、$D_m$ 检测器、反相复位）已整体撤除；原因是回声只活一拍（一周期改写 $99.3\%$），
        而检测至少需要一个周期，oracle 上界也只有 $1.82$ 倍。其脚本与数据只作历史证据保留。
  \item \textbf{边界}：见 §\ref{sec:limit}——参数口径不可互引、类 fal 的稳态代价依赖工作点、随机噪声不在确定性框架内、台架未测。
\end{enumerate}
